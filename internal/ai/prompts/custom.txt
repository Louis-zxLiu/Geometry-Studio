     '''
      import matplotlib as mpl
      import matplotlib.pyplot as plt
      import numpy as np
      from mpl_toolkits.mplot3d.art3d import Poly3DCollection
      from matplotlib.widgets import Slider, Button, CheckButtons

      mpl.rcParams['lines.antialiased'] = True
      mpl.rcParams['patch.antialiased'] = True
      # 默认旋转参数需要遵守
      mpl.rcParams["axes3d.mouserotationstyle"] = "azel"

      plt.rcParams.update({
          "font.family": "SimSun", 
          "mathtext.fontset": "stix", 
          "axes.unicode_minus": False, 
          "font.size": 11
      })

      MORANDI = {
          'plane_alpha': '#D0C8C6',  # 灰粉 
          'plane_beta': '#B5BDB5',   # 豆绿 
          'plane_gamma': '#A8B1C2',  # 雾蓝 
          'line_main': '#333333',    # 深灰 
          'line_aux': '#999999',     # 浅灰 
      }

      def calc_intersection(theta_deg):
          """计算动平面绕横轴旋转切割单位正方体的截面"""
          theta = np.radians(theta_deg)
          t = np.tan(theta)
          if t <= 1:
              return [(0, 0, 0), (1, 0, 0), (1, 1, t), (0, 1, t)], np.sqrt(1 + t ** 2)
          return [(0, 0, 0), (1, 0, 0), (1, 1/t, 1), (0, 1/t, 1)], np.sqrt((1/t) ** 2 + 1)

      fig = plt.figure(figsize=(10, 5), facecolor='white')

      # [左图]: 3D 正交视图
      ax1 = fig.add_axes([0.02, 0.22, 0.45, 0.7], projection='3d', proj_type='ortho')
      ax1.set_axis_off()
      ax1.view_init(elev=20, azim=-45)
      ax1.set_xlim(0, 1); ax1.set_ylim(0, 1); ax1.set_zlim(0, 1)
      ax1.text2D(0.5, 0.95, "空间几何: 动平面截面演化", transform=ax1.transAxes, ha='center', fontsize=13)

      # [右图]: 2D 曲线
      ax2 = fig.add_axes([0.55, 0.25, 0.38, 0.6])
      ax2.spines['top'].set_visible(False)
      ax2.spines['right'].set_visible(False)
      ax2.spines['left'].set_color(MORANDI['line_aux'])
      ax2.spines['bottom'].set_color(MORANDI['line_aux'])
      ax2.set_title("截面积曲线 $S(\\theta)$", pad=15, fontsize=13)
      ax2.set_xlabel("$\\theta (^{\\circ})$", color=MORANDI['line_main'])


      cube_lines = []
      for p in [0, 1]:
          cube_lines.append(ax1.plot([0, 1], [p, p], [0, 0], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
          cube_lines.append(ax1.plot([0, 1], [p, p], [1, 1], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
          cube_lines.append(ax1.plot([p, p], [0, 1], [0, 0], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
          cube_lines.append(ax1.plot([p, p], [0, 1], [1, 1], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
          cube_lines.append(ax1.plot([p, p], [0, 0], [0, 1], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
          cube_lines.append(ax1.plot([p, p], [1, 1], [0, 1], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])

      # 初始化截面、2D 曲线与面积填充
      init_verts, init_area = calc_intersection(30)
      poly3d = Poly3DCollection([init_verts], facecolors=MORANDI['plane_alpha'], edgecolors=MORANDI['line_main'], linewidths=1.2, alpha=0.4)
      ax1.add_collection3d(poly3d)

      thetas = np.linspace(0, 89.9, 200)
      areas = [calc_intersection(th)[1] for th in thetas]
      ax2.plot(thetas, areas, color=MORANDI['plane_gamma'], lw=1.2)
      dot2d, = ax2.plot([30], [init_area], 'o', color=MORANDI['line_main'], ms=5)

      # 使用经典通用的 fill_between 填充阴影面积
      fill_area = ax2.fill_between(thetas, areas, where=(thetas <= 30), color=MORANDI['plane_gamma'], alpha=0.15)

      # 4. 布局交互 UI 组件
      ax_slider = fig.add_axes([0.15, 0.08, 0.40, 0.02])  
      ax_chk    = fig.add_axes([0.62, 0.05, 0.15, 0.08])  
      ax_btn    = fig.add_axes([0.82, 0.06, 0.08, 0.05])  

      # 滑块配置
      slider = Slider(ax_slider, '旋转角 $\\theta$', 0, 89.9, valinit=30, color=MORANDI['plane_beta'], track_color='#EFEFEF')
      ax_slider.set_axis_off()
      slider.valtext.set_color(MORANDI['line_main'])

      # 复选框配置：使用通用、安全的传统 API 避坑
      chk = CheckButtons(ax_chk, ['辅助框线'], [True])
      ax_chk.set_axis_off()

      # 通过 ax_chk 内部的线条和图形对象直接、安全地修改样式
      for patch in ax_chk.patches:
          patch.set_edgecolor(MORANDI['line_aux'])
          patch.set_linewidth(0.8)
          patch.set_facecolor('white')
      for line in ax_chk.lines:
          line.set_color(MORANDI['line_main'])
          line.set_linewidth(1.2)

      # 重置按钮配置
      btn = Button(ax_btn, '重置', color='white', hovercolor='#F5F5F5')
      for spine in ax_btn.spines.values(): 
          spine.set_color(MORANDI['line_aux'])
          spine.set_linewidth(0.8)
      btn.label.set_color(MORANDI['line_main'])

      # 紧凑的高效回调逻辑
      def update(val):
          theta = slider.val
          verts, area = calc_intersection(theta)
          poly3d.set_verts([verts])
          dot2d.set_data([theta], [area])
          
          # 动态刷新阴影：先移除旧的，再一行 fill_between 重新绘制
          global fill_area
          fill_area.remove()
          fill_area = ax2.fill_between(thetas, areas, where=(thetas <= theta), color=MORANDI['plane_gamma'], alpha=0.15)
          fig.canvas.draw_idle()

      def toggle_visibility(label):
          state = chk.get_status()[0]
          for line in cube_lines:
              line.set_visible(state)
          fig.canvas.draw_idle()

      slider.on_changed(update)
      chk.on_clicked(toggle_visibility)
      btn.on_clicked(lambda event: slider.reset())

      plt.show()
