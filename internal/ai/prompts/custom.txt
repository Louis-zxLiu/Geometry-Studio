      import matplotlib.pyplot as plt
      import numpy as np
      from mpl_toolkits.mplot3d.art3d import Poly3DCollection

      plt.rcParams['axes.unicode_minus'] = False
      plt.rcParams.update({
      "text.usetex": False,
      "font.family": "SimSun",
      "mathtext.fontset": "stix",
      "axes.unicode_minus": False,
      "font.size": 12
        }) 
      # 字体必须遵守如上规定

      # ================= 配置参数 =================
      FIG_SIZE = (10, 4.5)
      DPI = 900
      LINE_WIDTH = 1.2

      MORANDI = {
          'plane_alpha': '#D0C8C6',  # 灰粉
          'plane_beta': '#B5BDB5',  # 豆绿
          'plane_gamma': '#A8B1C2',  # 雾蓝
          'line_main': '#333333',  # 深灰
          'line_aux': '#999999',  # 浅灰
      }

      plt.rcParams.update({
          "text.usetex": False,
          "mathtext.fontset": "stix",
          "font.family": "serif",
          "font.size": 14
      })


      # ================= 辅助函数 =================
      def create_figure():
          fig = plt.figure(figsize=FIG_SIZE, dpi=DPI)
          fig.patch.set_facecolor('white')
          ax1 = fig.add_subplot(121, projection='3d', proj_type='ortho')
          ax2 = fig.add_subplot(122, projection='3d', proj_type='ortho')
          for ax in [ax1, ax2]:
              ax.set_axis_off()
              ax.view_init(elev=20, azim=-55)
              ax.set_xlim([0, 1])
              ax.set_ylim([0, 1])
              ax.set_zlim([0, 1])
          return fig, ax1, ax2


      def draw_plane(ax, vertices, color, alpha=0.35, label='', label_pos=None):
          poly = Poly3DCollection([vertices], facecolors=color, edgecolors=MORANDI['line_aux'],
                                  linewidths=0.8, alpha=alpha, zorder=1)
          ax.add_collection3d(poly)
          if label and label_pos:
              ax.text(*label_pos, f"${label}$", color='#444444', style='italic', fontsize=18, zorder=5)


      def draw_line(ax, p1, p2, color=MORANDI['line_main'], ls='-', label='', label_pos=None):
          ax.plot([p1[0], p2[0]], [p1[1], p2[1]], [p1[2], p2[2]],
                  color=color, linewidth=LINE_WIDTH, linestyle=ls, zorder=4)
          if label and label_pos:
              ax.text(*label_pos, f"${label}$", color=color, style='italic', fontsize=18, zorder=5)


      def add_formula(ax, text):
          ax.text2D(0.5, -0.05, text, transform=ax.transAxes,
                    ha='center', va='top', fontsize=18, color='#222222')


      # ================= 核心绘图：面面平行 =================
      def plot_plane_plane_parallel():
          fig, ax1, ax2 = create_figure()

          # 空间方程基础: alpha 平面 z=0.2，beta 平面 z=0.6
          alpha_v = [(0.1, 0.1, 0.2), (0.9, 0.1, 0.2), (0.9, 0.9, 0.2), (0.1, 0.9, 0.2)]
          beta_v = [(0.1, 0.1, 0.6), (0.9, 0.1, 0.6), (0.9, 0.9, 0.6), (0.1, 0.9, 0.6)]

          # ---------------- 左图：判定定理 ----------------
          draw_plane(ax1, alpha_v, MORANDI['plane_alpha'], label='\\alpha', label_pos=(0.15, 0.85, 0.2))
          draw_plane(ax1, beta_v, MORANDI['plane_beta'], label='\\beta', label_pos=(0.15, 0.85, 0.6))

          # 在 beta 面内(z=0.6)严格取两条相交直线
          draw_line(ax1, (0.3, 0.3, 0.6), (0.7, 0.7, 0.6), label='a', label_pos=(0.75, 0.7, 0.6))
          draw_line(ax1, (0.3, 0.7, 0.6), (0.7, 0.3, 0.6), label='b', label_pos=(0.75, 0.3, 0.6))
          add_formula(ax1, r"$a \parallel \alpha, b \parallel \alpha, a \cap b = P \Rightarrow \alpha \parallel \beta$")

          # ---------------- 右图：性质定理 ----------------
          draw_plane(ax2, alpha_v, MORANDI['plane_alpha'], label='\\alpha', label_pos=(0.15, 0.85, 0.2))
          draw_plane(ax2, beta_v, MORANDI['plane_beta'], label='\\beta', label_pos=(0.15, 0.85, 0.6))

          # 截面 gamma 方程定为: z = y (过x轴且与y,z成45度角)
          gamma_v = [(0.1, 0.1, 0.1), (0.9, 0.1, 0.1), (0.9, 0.9, 0.9), (0.1, 0.9, 0.9)]
          draw_plane(ax2, gamma_v, MORANDI['plane_gamma'], label='\\gamma', label_pos=(0.1, 0.9, 0.95))

          # 交线 a 严格解: alpha(z=0.2) 与 gamma(z=y) 联立 -> y=0.2, z=0.2
          draw_line(ax2, (0.2, 0.2, 0.2), (0.8, 0.2, 0.2), label='a', label_pos=(0.85, 0.2, 0.2))

          # 交线 b 严格解: beta(z=0.6) 与 gamma(z=y) 联立 -> y=0.6, z=0.6
          draw_line(ax2, (0.2, 0.6, 0.6), (0.8, 0.6, 0.6), label='b', label_pos=(0.85, 0.6, 0.6))

          add_formula(ax2,
                      r"$\alpha \parallel \beta, \alpha \cap \gamma = a, \beta \cap \gamma = b \Rightarrow a \parallel b$")

          # 保存高精度图像
          fig.savefig('2_Plane_Plane_Parallel.png', bbox_inches='tight')
          plt.show()


        if __name__ == '__main__':
            plot_plane_plane_parallel()
            print("图2 (面面平行) 纯净版已生成！")
      '''
